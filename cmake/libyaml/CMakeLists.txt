cmake_minimum_required(VERSION 3.21)
project(cai_private_yaml VERSION 0.2.5 LANGUAGES C)
if(NOT EXISTS "${CAI_YAML_SOURCE}/include/yaml.h")
  message(FATAL_ERROR "CAI_YAML_SOURCE must name the verified libyaml source root")
endif()
set(YAML_VERSION_MAJOR 0)
set(YAML_VERSION_MINOR 2)
set(YAML_VERSION_PATCH 5)
set(YAML_VERSION_STRING "0.2.5")
configure_file("${CAI_YAML_SOURCE}/cmake/config.h.in" "include/config.h" @ONLY)
set(sources api dumper emitter loader parser reader scanner writer)
list(TRANSFORM sources PREPEND "${CAI_YAML_SOURCE}/src/")
list(TRANSFORM sources APPEND ".c")
add_library(yaml_objects OBJECT ${sources})
set_target_properties(yaml_objects PROPERTIES
  C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON)
target_include_directories(yaml_objects PRIVATE
  "${CAI_YAML_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}/include")
target_compile_definitions(yaml_objects PRIVATE HAVE_CONFIG_H YAML_DECLARE_STATIC _POSIX_C_SOURCE=200809L)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(yaml_objects PRIVATE -Werror=implicit-function-declaration)
endif()
add_library(yaml_static STATIC $<TARGET_OBJECTS:yaml_objects>)
add_library(yaml_shared SHARED $<TARGET_OBJECTS:yaml_objects>)
set_target_properties(yaml_static yaml_shared PROPERTIES OUTPUT_NAME yaml)
set_target_properties(yaml_shared PROPERTIES VERSION 0.2.5 SOVERSION 0)
install(TARGETS yaml_static yaml_shared ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(FILES "${CAI_YAML_SOURCE}/include/yaml.h" DESTINATION include)
install(FILES "${CAI_YAML_SOURCE}/License" DESTINATION share/doc/libyaml)
