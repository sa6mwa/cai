cmake_minimum_required(VERSION 3.21)
project(yaml_producer_fixture LANGUAGES C)
set(CAI_TARGET_ID fixture-native)
set(CAI_DEPS_DIR "${FIXTURE_DEPS}")
# This fixture tests build ownership/invalidation with a prepared, immutable
# prerequisite. Archive acquisition is outside this fixture's scope.
function(cai_dependency_archive_path name sha256 output)
  set(${output} "${FIXTURE_ARCHIVE}" PARENT_SCOPE)
endfunction()
function(cai_download_tarball name url archive prefix sha256 marker)
  file(SHA256 "${archive}" actual)
  if(NOT actual STREQUAL sha256 OR NOT EXISTS "${marker}")
    message(FATAL_ERROR "Fixture libyaml prerequisite is missing or has the wrong identity")
  endif()
endfunction()
include("${FIXTURE_MODULE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/consumer.c"
  "#include <yaml.h>\nint main(void) { return yaml_get_version_string() == 0; }\n")
add_executable(consumer "${CMAKE_CURRENT_BINARY_DIR}/consumer.c")
target_link_libraries(consumer PRIVATE cai_yaml_static)
if(CAI_ENABLE_COVERAGE)
  target_compile_options(consumer PRIVATE --coverage)
  target_link_options(consumer PRIVATE --coverage)
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/producer-path.txt" "${CAI_LIBYAML_BUILD_DIR}")
