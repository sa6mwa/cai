cmake_minimum_required(VERSION 3.21)
project(cai_luarocks_binding LANGUAGES C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CAI REQUIRED IMPORTED_TARGET cai)
pkg_check_modules(PSLOG REQUIRED IMPORTED_TARGET pslog)
pkg_check_modules(CAI_APP QUIET IMPORTED_TARGET cai-app)
if(NOT EXISTS "${CAI_LUA_INCLUDE_DIR}/lua.h" OR NOT EXISTS "${CAI_PSLOG_LUA_INCLUDE_DIR}/pslog_lua.h")
  message(FATAL_ERROR "Lua 5.5 and lua-pslog headers are required")
endif()
get_filename_component(cai_source_root "${CMAKE_CURRENT_LIST_DIR}" DIRECTORY)
include("${cai_source_root}/cmake/cai-export-policy.cmake")
add_library(cai_lua MODULE cai_lua.c)
set_target_properties(cai_lua PROPERTIES OUTPUT_NAME cai PREFIX ""
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(CAI_LUA_EXTENSION)
  set_target_properties(cai_lua PROPERTIES SUFFIX ".${CAI_LUA_EXTENSION}")
endif()
target_include_directories(cai_lua PRIVATE "${CAI_LUA_INCLUDE_DIR}" "${CAI_PSLOG_LUA_INCLUDE_DIR}")
target_link_libraries(cai_lua PRIVATE PkgConfig::CAI PkgConfig::PSLOG)
target_compile_definitions(cai_lua PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(cai_lua PRIVATE -std=c89 -Wall -Wextra -Werror
  -Wpedantic -Wno-long-long -Wstrict-prototypes -Wmissing-prototypes
  -Wshadow -Wpointer-arith -Wcast-qual -Wwrite-strings -Wdeclaration-after-statement)
cai_apply_export_policy(cai_lua "${CMAKE_CURRENT_LIST_DIR}/cai_lua.exports")
if(CAI_APP_FOUND)
  add_library(cai_app_lua MODULE app_lua.c)
  set_target_properties(cai_app_lua PROPERTIES OUTPUT_NAME app PREFIX ""
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/cai")
  target_include_directories(cai_app_lua PRIVATE "${CAI_LUA_INCLUDE_DIR}" "${CAI_PSLOG_LUA_INCLUDE_DIR}")
  target_link_libraries(cai_app_lua PRIVATE PkgConfig::CAI_APP ${CMAKE_DL_LIBS})
  target_compile_options(cai_app_lua PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic -Wno-long-long)
  target_link_options(cai_app_lua PRIVATE -Wl,--allow-shlib-undefined)
  cai_apply_export_policy(cai_app_lua "${CMAKE_CURRENT_LIST_DIR}/cai_app.exports")
  install(TARGETS cai_app_lua LIBRARY DESTINATION lib/lua/5.5/cai)
endif()
if(APPLE)
  target_link_options(cai_lua PRIVATE -undefined dynamic_lookup)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(cai_lua PRIVATE -Wl,--allow-shlib-undefined)
  target_link_libraries(cai_lua PRIVATE ${CMAKE_DL_LIBS})
endif()
